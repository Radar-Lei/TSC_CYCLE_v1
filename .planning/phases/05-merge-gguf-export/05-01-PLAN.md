import gguf
from tokenizers import Tokenizer, models, pre_tokenizers, decoders

reader = gguf.GGUFReader("runs/20260507T032419Z/gguf/model.q4_K_M.gguf")
# 已知字段名（probe 步骤验证过）：
#   tokenizer.ggml.model    -> "gpt2" 或 "llama" (BPE family)
#   tokenizer.ggml.tokens   -> list[bytes] (vocab，索引即 token id)
#   tokenizer.ggml.merges   -> list[bytes] (BPE merges, 每条 "AA BB" 字节串)
#   tokenizer.ggml.pre      -> pre-tokenizer 名（qwen2 / llama-bpe 等）
fields = {f.name: f for f in reader.fields}
tokens = [bytes(fields["tokenizer.ggml.tokens"].parts[i]).decode("utf-8", errors="replace")
          for i in fields["tokenizer.ggml.tokens"].data]
merges = [bytes(fields["tokenizer.ggml.merges"].parts[i]).decode("utf-8", errors="replace")
          for i in fields["tokenizer.ggml.merges"].data]
vocab = {tok: i for i, tok in enumerate(tokens)}
merges_pairs = [tuple(m.split(" ", 1)) for m in merges if " " in m]
bpe = Tokenizer(models.BPE(vocab=vocab, merges=merges_pairs, fuse_unk=False))
bpe.pre_tokenizer = pre_tokenizers.ByteLevel(add_prefix_space=False, use_regex=True)
bpe.decoder = decoders.ByteLevel()
gguf_ids = bpe.encode("<start_working_out>", add_special_tokens=False).ids
