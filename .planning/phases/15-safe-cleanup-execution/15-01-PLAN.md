    '.env', '.venv', '.claude', '.pytest_cache', '.planning', 'reality.log', 'reality_test.log',

    'artifacts/v4', 'data', 'artifacts', 'runs',
}

def resolve_under_root(rel: str) -> Path:
    candidate = (ROOT / rel).resolve()
    candidate.relative_to(ROOT)
    return candidate

inv = json.loads(INVENTORY.read_text())
entries = inv['entries']
candidates = {
    entry['path']
    for entry in entries
    if entry.get('phase15_allowed') == 'archive_only'
    and entry.get('recommended_action') == 'archive_candidate'
}
if candidates != set(EXPECTED):
    raise SystemExit(f'archive allowlist mismatch: {sorted(candidates)}')
remove_candidates = [entry['path'] for entry in entries if entry.get('recommended_action') == 'remove_candidate']
if remove_candidates:
    raise SystemExit(f'remove_candidate entries are not allowed in Phase 15: {remove_candidates}')
if set(EXPECTED) & FORBIDDEN_EXACT:
    raise SystemExit('expected move set intersects forbidden paths')
ARCHIVE_ROOT.mkdir(parents=True, exist_ok=True)
for src_rel, dst_rel in EXPECTED.items():
    src = resolve_under_root(src_rel)
    dst = resolve_under_root(dst_rel)
    dst.relative_to(ARCHIVE_ROOT.resolve())
    if not src.exists():
        raise SystemExit(f'source missing before move: {src}')
    if not src.is_dir():
        raise SystemExit(f'source is not a directory: {src}')
    if dst.exists():
        raise SystemExit(f'destination already exists: {dst}')
    if src.stat().st_dev != dst.parent.stat().st_dev:
        raise SystemExit(f'not same filesystem: {src} -> {dst}')
for src_rel, dst_rel in EXPECTED.items():
    os.rename(resolve_under_root(src_rel), resolve_under_root(dst_rel))
print('archived:', ', '.join(EXPECTED))
PY
