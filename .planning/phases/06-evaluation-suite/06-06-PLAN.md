           filt = [r for r in rows if r["backend"]==backend and r["split_hint"]=="ood" and not r["trivial"]]

           return sum(1 for r in filt if r["lint_ok"])/len(filt), len(filt)
       hf_rate, hf_n = ood_lint_rate(rows, "hf_bf16")

       def ood_mae(rows, backend):
