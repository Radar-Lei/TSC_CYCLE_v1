    ```python
    from tsc_cycle.constraint_lint import validate, is_trivial

    def score_constraint(prediction_input: dict, solution: dict | None) -> dict:
        """Returns {"lint_ok": bool, "violations": list[str], "trivial": bool}.
        若 solution is None → lint_ok=False, violations=["unparseable"]."""
        trivial = is_trivial(prediction_input)
        if solution is None:
            return {"lint_ok": False, "violations": ["unparseable"], "trivial": trivial}
        res = validate(prediction_input, solution)
        return {"lint_ok": bool(res.ok),
                "violations": [v.value if hasattr(v, "value") else str(v) for v in (res.violations or [])],
                "trivial": trivial}
    ```

    **B. tsc_cycle/eval/metrics_mae.py**
    ```python
    def score_mae(student_solution: dict | None, teacher_solution: dict) -> dict:
        """Returns {"mae": float | None, "exact_match": bool, "n_phases": int, "per_phase_abs_err": list[int]}.
        - phase 配对：按 teacher_solution.keys() 顺序
        - 若 student_solution is None or any phase missing → mae=None, exact_match=False
        - 整数差 abs(int(student[k]) - int(teacher[k]))"""
        if student_solution is None:
            return {"mae": None, "exact_match": False,
                    "n_phases": len(teacher_solution), "per_phase_abs_err": []}
        keys = list(teacher_solution.keys())
        if not all(k in student_solution for k in keys):
            return {"mae": None, "exact_match": False,
                    "n_phases": len(keys), "per_phase_abs_err": []}
        errs = [abs(int(student_solution[k]) - int(teacher_solution[k])) for k in keys]
        return {"mae": float(sum(errs)) / len(errs),
                "exact_match": all(e == 0 for e in errs),
                "n_phases": len(keys), "per_phase_abs_err": errs}
    ```

    **C. tsc_cycle/eval/metrics_reasoning.py**
    ```python
    import re
    from tsc_cycle.prompt_builder import parse_assistant_output

    KEYWORDS = ["pred_saturation", "min_green", "max_green", "pred_wait"]

    def score_reasoning(raw_text: str, prediction_input: dict) -> dict:
        """Rule-based reasoning quality:
        - extract reasoning segment via parse_assistant_output (first element)
        - count occurrences of phase_id strings + min_green/max_green numbers + KEYWORDS
        - tier: 0 hits=miss, 1-2=partial, >=3=full
        Returns {"reasoning_tier": "miss"|"partial"|"full", "hit_count": int,
                 "keywords_found": list[str], "numbers_found": list[int]}.
        """
        reasoning, _ = parse_assistant_output(raw_text)
        if not reasoning:
            return {"reasoning_tier": "miss", "hit_count": 0,
                    "keywords_found": [], "numbers_found": []}
        kws = [k for k in KEYWORDS if k in reasoning]
        # 抽取 input 的 min_green / max_green 数字（容忍 phases dict 或 phase_min_green/max_green 两种 schema）
        numbers = []
        phases = prediction_input.get("phases") or {}
        if phases:
            for pid, p in phases.items():
                for fld in ("min_green", "max_green"):
                    v = p.get(fld) if isinstance(p, dict) else None
                    if isinstance(v, int):
                        numbers.append(v)
        else:
            for fld in ("phase_min_green", "phase_max_green"):
                d = prediction_input.get(fld) or {}
                for v in d.values():
                    if isinstance(v, int):
                        numbers.append(v)
        nums_in_reasoning = [n for n in numbers if re.search(rf"\b{n}\b", reasoning)]
        hits = len(kws) + len(nums_in_reasoning)
        tier = "full" if hits >= 3 else ("partial" if hits >= 1 else "miss")
        return {"reasoning_tier": tier, "hit_count": hits,
                "keywords_found": kws, "numbers_found": nums_in_reasoning}
    ```

    **D. tsc_cycle/eval/metrics_ood_gap.py**
    ```python
    def compute_ood_gap(per_sample: list[dict], backend: str, metric_key: str) -> dict:
        """Aggregate per_sample list (already filtered to one backend) into id/ood splits.
        per_sample row schema: {sample_id, backend, split_hint, lint_ok, mae, exact_match, ...}
        - rate metrics (lint_ok / exact_match): mean of bool
        - mae: mean of non-None mae values
        Returns {"backend": ..., "metric": metric_key, "id": float, "ood": float, "gap": float}.
        gap = id - ood (positive => OOD degradation)."""
        rows = [r for r in per_sample if r["backend"] == backend]
        def agg(rows_, key):
            if key in ("lint_ok", "exact_match"):
                return sum(1 for r in rows_ if r.get(key)) / max(1, len(rows_))
            elif key == "mae":
                vals = [r["mae"] for r in rows_ if r.get("mae") is not None]
                return sum(vals) / max(1, len(vals)) if vals else float("nan")
            else:
                raise ValueError(key)
        id_rows = [r for r in rows if r["split_hint"] == "id"]
        ood_rows = [r for r in rows if r["split_hint"] == "ood"]
        id_v = agg(id_rows, metric_key)
        ood_v = agg(ood_rows, metric_key)
        return {"backend": backend, "metric": metric_key,
                "id": id_v, "ood": ood_v, "gap": id_v - ood_v}
    ```
